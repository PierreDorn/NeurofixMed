'use client';

import { useEffect, useRef, useState, useTransition } from 'react';
import {
  enviarMensagem,
  gerarFlashcards,
  gerarResumo,
  gerarQuestoes,
  gerarKitCompleto,
} from './actions';

type Role = 'user' | 'assistant';
type Msg = { id: string; role: Role; content: string };
type Acao = 'flashcards' | 'resumo' | 'questoes' | 'kit' | null;

const ACOES: { id: Exclude<Acao, null>; label: string; icon: string; desc: string }[] = [
  { id: 'flashcards', label: 'Criar flashcards', icon: '🔁', desc: '10 cards prontos pra revisar' },
  { id: 'resumo',     label: 'Criar resumo',     icon: '📝', desc: 'resumo estruturado em markdown' },
  { id: 'questoes',   label: 'Criar questões',   icon: '❓', desc: '5 questões estilo residência' },
  { id: 'kit',        label: 'Criar kit completo', icon: '🎯', desc: 'resumo + flashcards + questões' },
];

function uid() {
  return Math.random().toString(36).slice(2, 10) + Date.now().toString(36);
}

export default function IAChat() {
  const [mensagens, setMensagens] = useState<Msg[]>([]);
  const [input, setInput] = useState('');
  const [acaoAtiva, setAcaoAtiva] = useState<Acao>(null);
  const [temaAcao, setTemaAcao] = useState('');
  const [erro, setErro] = useState<string | null>(null);
  const [pending, startTransition] = useTransition();
  const scrollRef = useRef<HTMLDivElement>(null);
  const inputRef = useRef<HTMLTextAreaElement>(null);

  useEffect(() => {
    scrollRef.current?.scrollTo({ top: scrollRef.current.scrollHeight, behavior: 'smooth' });
  }, [mensagens, pending]);

  function adicionar(role: Role, content: string) {
    setMensagens(prev => [...prev, { id: uid(), role, content }]);
  }

  function enviar() {
    const texto = input.trim();
    if (!texto || pending) return;
    setInput('');
    setErro(null);
    adicionar('user', texto);
    const historico = mensagens.map(m => ({ role: m.role, content: m.content }));
    startTransition(async () => {
      const res = await enviarMensagem(historico, texto);
      if (res.ok) adicionar('assistant', res.content);
      else setErro(res.error);
    });
  }

  function executarAcao() {
    const tema = temaAcao.trim();
    if (!tema || !acaoAtiva || pending) return;
    const label =
      acaoAtiva === 'flashcards' ? 'flashcards' :
      acaoAtiva === 'resumo'     ? 'um resumo' :
      acaoAtiva === 'questoes'   ? 'questões' :
                                   'um kit completo';
    adicionar('user', `Crie ${label} sobre: ${tema}`);
    setErro(null);
    const acao = acaoAtiva;
    setAcaoAtiva(null);
    setTemaAcao('');
    startTransition(async () => {
      const res =
        acao === 'flashcards' ? await gerarFlashcards(tema) :
        acao === 'resumo'     ? await gerarResumo(tema) :
        acao === 'questoes'   ? await gerarQuestoes(tema) :
                                await gerarKitCompleto(tema);
      if (res.ok) adicionar('assistant', res.content);
      else setErro(res.error);
    });
  }

  function onKeyDown(e: React.KeyboardEvent<HTMLTextAreaElement>) {
    if (e.key === 'Enter' && !e.shiftKey) {
      e.preventDefault();
      enviar();
    }
  }

  return (
    <div className="ia-shell">
      <style>{CSS}</style>

      <header className="ia-header">
        <div>
          <h1>Estudar com IA</h1>
          <p>Tutor NeuroFix Med — pergunte qualquer coisa de medicina</p>
        </div>
        {mensagens.length > 0 && (
          <button className="btn-ghost ia-clear" onClick={() => { setMensagens([]); setErro(null); }}>
            Nova conversa
          </button>
        )}
      </header>

      <div ref={scrollRef} className="ia-stream">
        {mensagens.length === 0 ? (
          <div className="ia-empty">
            <div className="ia-empty-icon">🧠</div>
            <h2>Como posso ajudar nos seus estudos hoje?</h2>
            <p>Pergunte sobre qualquer tema de medicina ou use um dos botões abaixo pra gerar material de estudo pronto.</p>
          </div>
        ) : (
          mensagens.map(m => (
            <div key={m.id} className={`ia-msg ${m.role}`}>
              <div className="ia-avatar">{m.role === 'user' ? 'EU' : 'NF'}</div>
              <div className="ia-bubble">
                <pre>{m.content}</pre>
              </div>
            </div>
          ))
        )}

        {pending && (
          <div className="ia-msg assistant">
            <div className="ia-avatar">NF</div>
            <div className="ia-bubble ia-typing">
              <span></span><span></span><span></span>
            </div>
          </div>
        )}

        {erro && (
          <div className="ia-erro">
            <strong>Erro:</strong> {erro}
          </div>
        )}
      </div>

      <footer className="ia-footer">
        {acaoAtiva ? (
          <div className="ia-acao-form">
            <label>
              Tema para {ACOES.find(a => a.id === acaoAtiva)?.label.toLowerCase()}:
            </label>
            <div className="ia-acao-row">
              <input
                type="text"
                autoFocus
                placeholder="Ex: insuficiência cardíaca, diabetes mellitus tipo 2..."
                value={temaAcao}
                onChange={e => setTemaAcao(e.target.value)}
                onKeyDown={e => {
                  if (e.key === 'Enter') executarAcao();
                  if (e.key === 'Escape') { setAcaoAtiva(null); setTemaAcao(''); }
                }}
              />
              <button className="btn-primary" onClick={executarAcao} disabled={!temaAcao.trim() || pending}>
                Gerar
              </button>
              <button className="btn-ghost" onClick={() => { setAcaoAtiva(null); setTemaAcao(''); }}>
                Cancelar
              </button>
            </div>
          </div>
        ) : (
          <>
            <div className="ia-input-row">
              <textarea
                ref={inputRef}
                rows={1}
                placeholder="Pergunte algo de medicina... (Enter envia, Shift+Enter quebra linha)"
                value={input}
                onChange={e => setInput(e.target.value)}
                onKeyDown={onKeyDown}
                disabled={pending}
              />
              <button className="btn-primary ia-send" onClick={enviar} disabled={!input.trim() || pending}>
                Enviar
              </button>
            </div>

            <div className="ia-acoes">
              {ACOES.map(a => (
                <button
                  key={a.id}
                  className="ia-acao-btn"
                  onClick={() => setAcaoAtiva(a.id)}
                  disabled={pending}
                  title={a.desc}
                >
                  <span className="ia-acao-icon">{a.icon}</span>
                  <span>{a.label}</span>
                </button>
              ))}
            </div>
          </>
        )}
      </footer>
    </div>
  );
}
